match.ns

v3.0.0 
fn f(x: i64) -> i64 {
    match x {
        1 => {
            return 10;
        },
        2 => {
            return 20;
        },
        _ => {
            return 30;
        },
    }
    return 0;
}
fn bs(b: bool) -> i64 {
    match b {
        true => {
            return 1;
        },
        false => {
            return 0;
        },
    }
    return 9;
}
fn main() {
    print(f(1));
    print(f(2));
    print(f(9));
    print(bs(true));
    print(bs(false));
}
