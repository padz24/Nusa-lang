enum Shape {
    Circle(i64), Rect(i64, i64), Dot
}
fn tag(s: Shape) -> i64 {
    match s {
        Circle(r) => {
            return r;
        },
        Rect(w, h) => {
            return w + h;
        },
        Dot => {
            return 0 - 1;
        },
    }
    return 0;
}
fn main() {
    print(tag(Shape.Circle(5)));
    print(tag(Shape.Rect(3, 4)));
    print(tag(Shape.Dot));
}
