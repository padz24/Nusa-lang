v3.0.0 
fn apply(f: fn(i64) -> i64, x: i64) -> i64 {
    return f(x);
}
fn main() {
    let add = |a: i64, b: i64| a + b;
    print(add(1, 2));
    print(apply(|x: i64| x * 2, 21));
    let k = 10;
    let f = |x: i64| x + k;
    print(f(5));
    print(apply(|x| x + k, 1));
}
